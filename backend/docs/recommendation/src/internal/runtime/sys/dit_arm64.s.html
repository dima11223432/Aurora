TEXT ·EnableDIT(SB),$0-1
    MRS DIT, R0
    UBFX $24, R0, $1, R1
    MOVB R1, ret+0(FP)
    MSR $1, DIT

    // performant when available. We should detect its availability
    // and use it when we can.
    DSB $7  // nsh
    ISB $15 // sy
    RET

    RET

    RET
