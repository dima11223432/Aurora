func ( *bytes.Buffer,  rune) {
	const  = "0123456789abcdef"
	if !isWhitespace() {
		.WriteByte('\\')
		.WriteRune()
	} else {
		switch  {
		case '\b':
			.WriteString(`\backspace`)
		case '\f':
			.WriteString(`\formfeed`)
		case '\n':
			.WriteString(`\newline`)
		case '\r':
			.WriteString(`\return`)
		case '\t':
			.WriteString(`\tab`)
		case ' ':
			.WriteString(`\space`)
		default:
			.WriteByte('\\')
			.WriteByte('u')
			.WriteByte([>>12&0xF])

			.WriteByte([>>4&0xF])
			.WriteByte([&0xF])
