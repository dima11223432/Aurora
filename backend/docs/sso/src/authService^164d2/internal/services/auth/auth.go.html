// it automatically registers them. Returns a JWT token on success.

	)
	.Info("attempting to login user")

	,  := .userProvider.User(, .Telegram_id)

// RegisterNewUser creates a new user in the storage.
// Returns ErrUserExists if the user already exists.
func ( *Auth) ( context.Context,  models.User) (int64, error) {
	const  = "auth.RegisterNewUser"

	 := .log.With(
		slog.String("op", ),
		slog.Int64("telegram_id", .Telegram_id),
		slog.String("username", .Username),
		slog.String("first_name", .First_name),
	)

	.Info("attempting to register new user")

	,  := .userProvider.User(, .Telegram_id)
	if  == nil {
		.Warn("user already exists", slog.Int64("existing_id", .ID))
		return .ID, fmt.Errorf("%s: %w", , ErrUserExists)
	}
	if !errors.Is(, storage.ErrUserNotFound) {
		.Error("failed to check user existence", slog.String("error", .Error()))
		return 0, fmt.Errorf("%s: %w", , )
	}

	,  := .userSaver.SaveUser(, )
	if  != nil {
		.Error("failed to save user", slog.String("error", .Error()))
		return 0, fmt.Errorf("%s: %w", , )
	}

	.Info("user registered successfully",
		slog.Int64("user_id", ),
		slog.Int64("telegram_id", .Telegram_id),
	)
	return , nil
}
