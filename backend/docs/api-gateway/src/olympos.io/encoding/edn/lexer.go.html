type lexState int

const (
	lexCont    = lexState(iota) // continue reading
	lexIgnore                   // values you can ignore, just whitespace and comments atm
	lexEnd                      // value ended with input given in
	lexEndPrev                  // value ended with previous input
	lexError                    // erroneous input
)

type tokenType int

const ( // value types from lexer
	tokenSymbol = tokenType(iota)
	tokenKeyword
	tokenString
	tokenInt
	tokenFloat
	tokenTag
	tokenChar
	tokenListStart
	tokenListEnd
	tokenVectorStart
	tokenVectorEnd
	tokenMapStart
	tokenMapEnd
	tokenSetStart
	tokenDiscard

	tokenError
)

func ( tokenType) () string {
	switch  {
	case tokenSymbol:
		return "symbol"
	case tokenKeyword:
		return "keyword"
	case tokenString:
		return "string"
	case tokenInt:
		return "integer"
	case tokenFloat:
		return "float"
	case tokenTag:
		return "tag"
	case tokenChar:
		return "character"
	case tokenListStart:
		return "list start"
	case tokenListEnd:
		return "list end"
	case tokenVectorStart:
		return "vector start"
	case tokenVectorEnd:
		return "vector end"
	case tokenMapStart:
		return "map start"
	case tokenMapEnd:
		return "map/set end"
	case tokenSetStart:
		return "set start"
	case tokenDiscard:
		return "discard token"
	case tokenError:
		return "error"
	default:
		return "[unknown]"
	}
}

const tokenSetEnd = tokenMapEnd // sets ends the same way as maps do

// A SyntaxError is a description of an EDN syntax error.

	msg    string // description of error
	Offset int64  // error occurred after reading Offset bytes
}

func ( *SyntaxError) () string {
	return .msg
}

	}

		return true

type lexer struct {
	state    func(rune) lexState
	err      error
	position int64
	token    tokenType

	count     int    // counter is used in some functions within the lexer
	expecting []rune // expecting is used to avoid duplication when we expect e.g. \newline
}

func ( *lexer) () {
	.state = .stateBegin
	.token = tokenType(-1)
	.err = nil
}

func ( *lexer) () lexState {
	if .err != nil {
		return lexError
	}
	 := .state(' ')
	if  == lexCont {
		.err = &SyntaxError{"unexpected end of EDN input", .position}
		 = lexError
	}
	if .err != nil {
		return lexError
	}
	if  == lexEndPrev {
		return lexEnd
	}
	return 
}

func ( *lexer) ( rune) lexState {
	switch {
	case isWhitespace():
		return lexIgnore
	case  == '{':
		.token = tokenMapStart
		return lexEnd
	case  == '}':
		.token = tokenMapEnd
		return lexEnd
	case  == '[':
		.token = tokenVectorStart
		return lexEnd
	case  == ']':
		.token = tokenVectorEnd
		return lexEnd
	case  == '(':
		.token = tokenListStart
		return lexEnd
	case  == ')':
		.token = tokenListEnd
		return lexEnd
	case  == '#':
		.state = .statePound
		return lexCont
	case  == ':':
		.state = .stateKeyword
		return lexCont
	case  == '/': // ohh, the lovely slash edge case
		.token = tokenSymbol
		.state = .stateEndLit
		return lexCont
	case  == '+':
		.state = .statePos
		return lexCont
	case  == '-':
		.state = .stateNeg
		return lexCont
	case  == '.':
		.token = tokenSymbol
		.state = .stateDotPre
		return lexCont
	case  == '"':
		.state = .stateInString
		return lexCont
	case  == '\\':
		.state = .stateChar
		return lexCont
	case okSymbolFirst() || u.IsLetter():
		.token = tokenSymbol
		.state = .stateSym
		return lexCont
	case '0' <  &&  <= '9':
		.state = .state1
		return lexCont
	case  == '0':
		.state = .state0
		return lexCont
	case  == ';':
		.state = .stateComment
		return lexIgnore
	}
	return .error(, "- unexpected rune")
}

func ( *lexer) ( rune) lexState {
	if  == '\n' {
		.state = .stateBegin
	}
	return lexIgnore
}

func ( *lexer) ( rune) lexState {
	if isWhitespace() ||  == '"' ||  == '{' ||  == '[' ||  == '(' ||  == ')' ||  == ']' ||  == '}' ||  == '\\' ||  == ';' {
		return lexEndPrev
	}
	return .error(, "- unexpected rune after legal "+.token.String())
}

func ( *lexer) ( rune) lexState {
	switch {
	case  == ':':
		.state = .stateError
		.err = &SyntaxError{"EDN does not support namespace-qualified keywords", .position}
		return lexError
	case  == '/':
		.state = .stateError
		.err = &SyntaxError{"keywords cannot begin with /", .position}
		return lexError
	case okSymbol() || u.IsLetter() || ('0' <=  &&  <= '9'):
		.token = tokenKeyword
		.state = .stateSym
		return lexCont
	}
	return .error(, "after keyword start")
}

		.state = .
		return lexCont
	case  == '/':
		.state = .stateSlash
		return lexCont
	}
	return .stateEndLit()
}

// value is '0'
func ( *lexer) ( rune) lexState {
	switch {
	case  == '.':
		.state = .stateDot
		return lexCont
	case  == 'e' ||  == 'E':
		.state = .stateE
		return lexCont
	case  == 'M': // bigdecimal
		.token = tokenFloat
		.state = .stateEndLit
		return lexCont // must be ws or delimiter afterwards
	case  == 'N': // bigint
		.token = tokenInt
		.state = .stateEndLit
		return lexCont // must be ws or delimiter afterwards
	}
	.token = tokenInt
	return .stateEndLit()
}

// anything but a result starting with 0. example '10', '34'
func ( *lexer) ( rune) lexState {
	if '0' <=  &&  <= '9' {
		return lexCont
	}
	return .state0()
}

	case  == '/':

		.state = .stateSlash

	if '0' <=  &&  <= '9' {
		.state = .stateDot0
		return lexCont
	}
	// TODO (?): The spec says that there must be numbers after the dot, yet

	return .error(, "after decimal point in numeric literal")

// after reading numeric values plus '.', example: '12.34'
func ( *lexer) ( rune) lexState {
	switch {
	case '0' <=  &&  <= '9':
		return lexCont
	case  == 'e' ||  == 'E':
		.state = .stateE
		return lexCont
	case  == 'M':
		.token = tokenFloat
		.state = .stateEndLit
		return lexCont
	}
	.token = tokenFloat
	return .stateEndLit()
}

// stateE is the state after reading the mantissa and e in a number,
// such as after reading `314e` or `0.314e`.
func ( *lexer) ( rune) lexState {
	if  == '+' ||  == '-' {
		.state = .stateESign
		return lexCont
	}
	return .stateESign()
}

// stateESign is the state after reading the mantissa, e, and sign in a number,
// such as after reading `314e-` or `0.314e+`.
func ( *lexer) ( rune) lexState {
	if '0' <=  &&  <= '9' {
		.state = .stateE0
		return lexCont
	}
	return .error(, "in exponent of numeric literal")
}

// stateE0 is the state after reading the mantissa, e, optional sign,
// and at least one digit of the exponent in a number,
// such as after reading `314e-2` or `0.314e+1` or `3.14e0`.
func ( *lexer) ( rune) lexState {
	if '0' <=  &&  <= '9' {
		return lexCont
	}
	if  == 'M' {
		.token = tokenFloat
		.state = .stateEndLit
		return lexCont
	}
	.token = tokenFloat
	return .stateEndLit()
}

var (
	newlineRunes  = []rune("newline")
	returnRunes   = []rune("return")
	spaceRunes    = []rune("space")
	tabRunes      = []rune("tab")
	formfeedRunes = []rune("formfeed")
)

// stateChar after a backslash ('\')
func ( *lexer) ( rune) lexState {
	switch {
	// oh my, I'm so happy that none of these share the same prefix.
	case  == 'n':
		.count = 1
		.expecting = newlineRunes
		.state = .stateSpecialChar
		return lexCont
	case  == 'r':
		.count = 1
		.expecting = returnRunes
		.state = .stateSpecialChar
		return lexCont
	case  == 's':
		.count = 1
		.expecting = spaceRunes
		.state = .stateSpecialChar
		return lexCont
	case  == 't':
		.count = 1
		.expecting = tabRunes
		.state = .stateSpecialChar
		return lexCont
	case  == 'f':
		.count = 1
		.expecting = formfeedRunes
		.state = .stateSpecialChar
		return lexCont
	case  == 'u':
		.count = 0
		.state = .stateUnicodeChar
		return lexCont
	case isWhitespace():
		.state = .stateError
		.err = &SyntaxError{"backslash cannot be followed by whitespace", .position}
		return lexError
	}
	// default is single name character
	.token = tokenChar
	.state = .stateEndLit
	return lexCont
}

func ( *lexer) ( rune) lexState {
	if  == .expecting[.count] {
		.count++
		if .count == len(.expecting) {
			.token = tokenChar
			.state = .stateEndLit
			return lexCont
		}
		return lexCont
	}
	if .count != 1 {
		return .error(, "after start of special character")
	}
	// it is likely just a normal character, like 'n' or 't'
	.token = tokenChar
	return .stateEndLit()
}

func ( *lexer) ( rune) lexState {
	if '0' <=  &&  <= '9' || 'a' <=  &&  <= 'f' || 'A' <=  &&  <= 'F' {
		.count++
		if .count == 4 {
			.token = tokenChar
			.state = .stateEndLit
		}
		return lexCont
	}
	if .count != 0 {
		return .error(, "after start of unicode character")
	}
	// likely just '\u'
	.token = tokenChar
	return .stateEndLit()
}

// stateInString is the state after reading `"`.
func ( *lexer) ( rune) lexState {
	if  == '"' {
		.token = tokenString
		return lexEnd
	}
	if  == '\\' {
		.state = .stateInStringEsc
		return lexCont
	}
	return lexCont
}

// stateInStringEsc is the state after reading `"\` during a quoted string.
func ( *lexer) ( rune) lexState {
	switch  {
	case 'b', 'f', 'n', 'r', 't', '\\', '/', '"':
		.state = .stateInString
		return lexCont
	case 'u':
		.state = .stateInStringEscU
		.count = 0
		return lexCont
	}
	return .error(, "in string escape code")
}

// stateInStringEscU is the state after reading `"\u` and l.count elements in a
// quoted string.
func ( *lexer) ( rune) lexState {
	if '0' <=  &&  <= '9' || 'a' <=  &&  <= 'f' || 'A' <=  &&  <= 'F' {
		.count++
		if .count == 4 {
			.state = .stateInString
		}
		return lexCont
	}
	// numbers
	return .error(, "in \\u hexadecimal character escape")
}

// after reading the character '#'
func ( *lexer) ( rune) lexState {
	switch {
	case  == '_':
		.token = tokenDiscard
		return lexEnd
	case  == '{':
		.token = tokenSetStart
		return lexEnd
	case u.IsLetter():
		.token = tokenTag
		.state = .stateSym
		return lexCont
	}
	return .error(, `after token starting with "#"`)
}

func ( *lexer) ( rune) lexState {
	return lexError

// error records an error and switches to the error state.
func ( *lexer) ( rune,  string) lexState {
	.state = .stateError
	.err = &SyntaxError{"invalid character " + quoteRune() + " " + , .position}
	return lexError
}

// quoteRune formats r as a quoted rune literal
func ( rune) string {
	// special cases - different from quoted strings
	if  == '\'' {
		return `'\''`
	}
	if  == '"' {
		return `'"'`
	}

	// use quoted string with different quotation marks
	 := strconv.Quote(string())
	return "'" + [1:len()-1] + "'"
}
