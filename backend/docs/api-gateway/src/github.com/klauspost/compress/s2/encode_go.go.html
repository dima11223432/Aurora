// emitLiteral writes a literal chunk and returns the number of bytes written.
//
// It assumes that:
//
//	dst is long enough to hold the encoded bytes
//	0 <= len(lit) && len(lit) <= math.MaxUint32
func ( []byte) int {
	if len() == 0 {
		return 0
	}
	switch {
	case len() <= 60:
		return len() + 1
	case len() <= 1<<8:
		return len() + 2
	case len() <= 1<<16:
		return len() + 3
	case len() <= 1<<24:
		return len() + 4
	default:
		return len() + 5
	}
}
