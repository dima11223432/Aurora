func () (float64, error) {
	// Alternately, the undocumented proc_pidinfo(PROC_PIDLISTFDS) can be used to
	// return a list of open fds, but that requires a way to call C APIs.  The
	// benefits, however, include fewer system calls and not failing when at the
	// open file soft limit.

	if ,  := os.Open("/dev/fd");  != nil {
		return 0.0, 
	} else {
		defer .Close()

		// Avoid ReadDir(), as it calls stat(2) on each descriptor.  Not only is
		// that info not used, but KQUEUE descriptors fail stat(2), which causes
		// the whole method to fail.
		if ,  := .Readdirnames(0);  != nil {
			return 0.0, 
		} else {
			// Subtract 1 to ignore the open /dev/fd descriptor above.
			return float64(len() - 1), nil
		}
	}
}

func ( *processCollector) ( chan<- Metric) {
	if ,  := unix.SysctlKinfoProcSlice("kern.proc.pid", os.Getpid());  == nil {
		if len() == 1 {
			 := float64([0].Proc.P_starttime.Nano() / 1e9)
			 <- MustNewConstMetric(.startTime, GaugeValue, )
		} else {
			 = fmt.Errorf("sysctl() returned %d proc structs (expected 1)", len())
			.reportError(, .startTime, )
		}
	} else {
		.reportError(, .startTime, )
	}

	// The proc structure returned by kern.proc.pid above has an Rusage member,
	// but it is not filled in, so it needs to be fetched by getrusage(2).  For
	// that call, the UTime, STime, and Maxrss members are filled out, but not
	// Ixrss, Idrss, or Isrss for the memory usage.  Memory stats will require
	// access to the C API to call task_info(TASK_BASIC_INFO).
	 := unix.Rusage{}

	if  := unix.Getrusage(syscall.RUSAGE_SELF, &);  == nil {
		 := time.Duration(.Stime.Nano() + .Utime.Nano()).Seconds()
		 <- MustNewConstMetric(.cpuTotal, CounterValue, )
	} else {
		.reportError(, .cpuTotal, )
	}

	if ,  := getMemory();  == nil {
		 <- MustNewConstMetric(.rss, GaugeValue, float64(.rss))
		 <- MustNewConstMetric(.vsize, GaugeValue, float64(.vsize))
	} else if !errors.Is(, errNotImplemented) {
		// Don't report an error when support is not compiled in.
		.reportError(, .rss, )
		.reportError(, .vsize, )
	}

	if ,  := getOpenFileCount();  == nil {
		 <- MustNewConstMetric(.openFDs, GaugeValue, )
	} else {
		.reportError(, .openFDs, )
	}

	if ,  := getSoftLimit(syscall.RLIMIT_NOFILE);  == nil {
		 <- MustNewConstMetric(.maxFDs, GaugeValue, float64())
	} else {
		.reportError(, .maxFDs, )
	}

	if ,  := getSoftLimit(syscall.RLIMIT_AS);  == nil {
		 <- MustNewConstMetric(.maxVsize, GaugeValue, float64())
	} else {
		.reportError(, .maxVsize, )
	}

	// TODO: socket(PF_SYSTEM) to fetch "com.apple.network.statistics" might
	//  be able to get the per-process network send/receive counts.
}
