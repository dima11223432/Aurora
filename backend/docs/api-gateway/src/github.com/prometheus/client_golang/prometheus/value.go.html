// ValueType is an enumeration of metric types that represent a simple value.
type ValueType int

// Possible values for the ValueType enum. Use UntypedValue to mark a metric
// with an unknown type.
const (
	_ ValueType = iota
	CounterValue
	GaugeValue
	UntypedValue
)

var (
	CounterMetricTypePtr = func() *dto.MetricType {  := dto.MetricType_COUNTER; return & }()
	GaugeMetricTypePtr   = func() *dto.MetricType {  := dto.MetricType_GAUGE; return & }()
	UntypedMetricTypePtr = func() *dto.MetricType {  := dto.MetricType_UNTYPED; return & }()
)

func ( ValueType) () *dto.MetricType {
	switch  {
	case CounterValue:
		return CounterMetricTypePtr
	case GaugeValue:
		return GaugeMetricTypePtr
	default:
		return UntypedMetricTypePtr
	}
}
