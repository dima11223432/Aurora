type baseGoCollector struct {
	goroutinesDesc *Desc
	threadsDesc    *Desc
	gcDesc         *Desc
	gcLastTimeDesc *Desc
	goInfoDesc     *Desc
}

func () baseGoCollector {
	return baseGoCollector{
		goroutinesDesc: NewDesc(
			"go_goroutines",
			"Number of goroutines that currently exist.",
			nil, nil),
		threadsDesc: NewDesc(
			"go_threads",
			"Number of OS threads created.",
			nil, nil),
		gcDesc: NewDesc(
			"go_gc_duration_seconds",
			"A summary of the wall-time pause (stop-the-world) duration in garbage collection cycles.",
			nil, nil),
		gcLastTimeDesc: NewDesc(
			"go_memstats_last_gc_time_seconds",
			"Number of seconds since 1970 of last garbage collection.",
			nil, nil),
		goInfoDesc: NewDesc(
			"go_info",
			"Information about the Go environment.",
			nil, Labels{"version": runtime.Version()}),
	}
}

// Describe returns all descriptions of the collector.
func ( *baseGoCollector) ( chan<- *Desc) {
	 <- .goroutinesDesc
	 <- .threadsDesc
	 <- .gcDesc
	 <- .gcLastTimeDesc
	 <- .goInfoDesc
}

// Collect returns the current state of all metrics of the collector.
func ( *baseGoCollector) ( chan<- Metric) {
	 <- MustNewConstMetric(.goroutinesDesc, GaugeValue, float64(runtime.NumGoroutine()))

	 := getRuntimeNumThreads()
	 <- MustNewConstMetric(.threadsDesc, GaugeValue, )

	var  debug.GCStats
	.PauseQuantiles = make([]time.Duration, 5)
	debug.ReadGCStats(&)

	 := make(map[float64]float64)
	for ,  := range .PauseQuantiles[1:] {
		[float64(+1)/float64(len(.PauseQuantiles)-1)] = .Seconds()
	}
	[0.0] = .PauseQuantiles[0].Seconds()
	 <- MustNewConstSummary(.gcDesc, uint64(.NumGC), .PauseTotal.Seconds(), )
	 <- MustNewConstMetric(.gcLastTimeDesc, GaugeValue, float64(.LastGC.UnixNano())/1e9)
	 <- MustNewConstMetric(.goInfoDesc, GaugeValue, 1)
}

// memStatsMetrics provide description, evaluator, runtime/metrics name, and
// value type for memstat metrics.

	desc    *Desc
	eval    func(*runtime.MemStats) float64
	valType ValueType
}
