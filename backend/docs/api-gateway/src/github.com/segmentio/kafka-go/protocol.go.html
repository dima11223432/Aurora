func ( ApiVersion) ( fmt.State,  rune) {
	switch  {
	case 's':
		fmt.Fprint(, apiKey(.ApiKey))
	case 'd':
		switch {
		case .Flag('-'):
			fmt.Fprint(, .MinVersion)
		case .Flag('+'):
			fmt.Fprint(, .MaxVersion)
		default:
			fmt.Fprint(, .ApiKey)
		}
	case 'v':
		switch {
		case .Flag('-'):
			fmt.Fprintf(, "v%d", .MinVersion)
		case .Flag('+'):
			fmt.Fprintf(, "v%d", .MaxVersion)
		case .Flag('#'):
			fmt.Fprintf(, "kafka.ApiVersion{ApiKey:%d MinVersion:%d MaxVersion:%d}", .ApiKey, .MinVersion, .MaxVersion)
		default:
			fmt.Fprintf(, "%s[v%d:v%d]", apiKey(.ApiKey), .MinVersion, .MaxVersion)
		}
	}
}
