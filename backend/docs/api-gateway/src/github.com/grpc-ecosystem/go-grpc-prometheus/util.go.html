func ( *grpc.MethodInfo) grpcType {
	if !.IsClientStream && !.IsServerStream {
		return Unary
	}
	if .IsClientStream && !.IsServerStream {
		return ClientStream
	}
	if !.IsClientStream && .IsServerStream {
		return ServerStream
	}
	return BidiStream
}
