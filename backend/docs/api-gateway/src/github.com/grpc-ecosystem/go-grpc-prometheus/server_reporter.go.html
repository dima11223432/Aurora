package grpc_prometheus

type serverReporter struct {
	metrics     *ServerMetrics
	rpcType     grpcType
	serviceName string
	methodName  string
	startTime   time.Time
}

func ( *ServerMetrics,  grpcType,  string) *serverReporter {
	 := &serverReporter{
		metrics: ,
		rpcType: ,
	}
	if .metrics.serverHandledHistogramEnabled {
		.startTime = time.Now()
	}
	.serviceName, .methodName = splitMethodName()
	.metrics.serverStartedCounter.WithLabelValues(string(.rpcType), .serviceName, .methodName).Inc()
	return 
}

func ( *serverReporter) () {
	.metrics.serverStreamMsgReceived.WithLabelValues(string(.rpcType), .serviceName, .methodName).Inc()
}

func ( *serverReporter) () {
	.metrics.serverStreamMsgSent.WithLabelValues(string(.rpcType), .serviceName, .methodName).Inc()
}

func ( *serverReporter) ( codes.Code) {
	.metrics.serverHandledCounter.WithLabelValues(string(.rpcType), .serviceName, .methodName, .String()).Inc()
	if .metrics.serverHandledHistogramEnabled {
		.metrics.serverHandledHistogram.WithLabelValues(string(.rpcType), .serviceName, .methodName).Observe(time.Since(.startTime).Seconds())
	}
}
