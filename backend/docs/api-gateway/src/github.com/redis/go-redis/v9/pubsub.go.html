type Subscription struct {
	// Can be "subscribe", "unsubscribe", "psubscribe" or "punsubscribe".
	Kind string
	// Channel name we have subscribed to.
	Channel string
	// Number of channels we are currently subscribed to.
	Count int
}

func ( *Subscription) () string {
	return fmt.Sprintf("%s: %s", .Kind, .Channel)
}

// ReceiveTimeout acts like Receive but returns an error if message
// is not received in time. This is low-level API and in most cases
// Channel should be used instead.
func ( *PubSub) ( context.Context,  time.Duration) (interface{}, error) {

		.cmd = NewCmd()
	}

			// Log the error but don't fail the command execution
			// Push notification processing errors shouldn't break normal Redis operations
			internal.Logger.Printf(, "push: conn[%d] error processing pending notifications before reading reply: %v", .GetID(), )
		}
		return .cmd.readReply()
	})
	.releaseConnWithLock(, , ,  > 0)

	if  != nil {
		return nil, 
	}

	return .newMessage(, , .cmd.Val())
}

// Channel returns a Go channel for concurrently receiving messages.
// The channel is closed together with the PubSub. If the Go channel
// is blocked full for 1 minute the message is dropped.
// Receive* APIs can not be used after channel is created.
//
// go-redis periodically sends ping messages to test connection health
// and re-subscribes if ping can not received for 1 minute.
