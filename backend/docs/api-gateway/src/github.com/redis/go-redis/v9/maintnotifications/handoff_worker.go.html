// onDemandWorker processes handoff requests and exits when idle
func ( *handoffWorkerManager) () {

		}

		.activeWorkers.Add(-1)
		.workerWg.Done()
	}()

	// Create reusable timer to prevent timer leaks
	 := time.NewTimer(.workerTimeout)
	defer .Stop()

	for {
		// Reset timer for next iteration
		if !.Stop() {
			select {
			case <-.C:
			default:
			}
		}
		.Reset(.workerTimeout)

		select {
		case <-.shutdown:
			if internal.LogLevel.InfoOrAbove() {
				internal.Logger.Printf(context.Background(), logs.WorkerExitingDueToShutdown())
			}
			return
		case <-.C:
			// Worker has been idle for too long, exit to save resources
			if internal.LogLevel.InfoOrAbove() {
				internal.Logger.Printf(context.Background(), logs.WorkerExitingDueToInactivityTimeout(.workerTimeout))
			}
			return
		case  := <-.handoffQueue:
			// Check for shutdown before processing
			select {
			case <-.shutdown:
				if internal.LogLevel.InfoOrAbove() {
					internal.Logger.Printf(context.Background(), logs.WorkerExitingDueToShutdownWhileProcessing())
				}
				// Clean up the request before exiting
				.pending.Delete(.ConnID)
				return
			default:
				// Process the request
				.processHandoffRequest()
			}
		}
	}
}

// queueHandoff queues a handoff request for processing
// if err is returned, connection will be removed from pool
func ( *handoffWorkerManager) ( *pool.Conn) error {
	// Get handoff info atomically to prevent race conditions
	, ,  := .GetHandoffInfo()

	// on retries the connection will not be marked for handoff, but it will have retries > 0
	// if shouldHandoff is false and retries is 0, then we are not retrying and not do a handoff
	if ! && .HandoffRetries() == 0 {
		if internal.LogLevel.InfoOrAbove() {
			internal.Logger.Printf(context.Background(), logs.ConnectionNotMarkedForHandoff(.GetID()))
		}
		return errors.New(logs.ConnectionNotMarkedForHandoffError(.GetID()))
	}

	// Create handoff request with atomically retrieved data
	 := HandoffRequest{
		Conn:     ,
		ConnID:   .GetID(),
		Endpoint: ,
		SeqID:    ,
		Pool:     .poolHook.pool, // Include pool for connection removal on failure
	}

	select {
	// priority to shutdown
	case <-.shutdown:
		return ErrShutdown
	default:
		select {
		case <-.shutdown:
			return ErrShutdown
		case .handoffQueue <- :
			// Store in pending map
			.pending.Store(.ConnID, .SeqID)
			// Ensure we have a worker to process this request
			.ensureWorkerAvailable()
			return nil
		default:
			select {
			case <-.shutdown:
				return ErrShutdown
			case .handoffQueue <- :
				// Store in pending map
				.pending.Store(.ConnID, .SeqID)
				// Ensure we have a worker to process this request
				.ensureWorkerAvailable()
				return nil
			case <-time.After(100 * time.Millisecond): // give workers a chance to process
				// Queue is full - log and attempt scaling
				 := len(.handoffQueue)
				 := cap(.handoffQueue)
				if internal.LogLevel.WarnOrAbove() {
					internal.Logger.Printf(context.Background(), logs.HandoffQueueFull(, ))
				}
			}
		}
	}

	// Ensure we have workers available to handle the load
	.ensureWorkerAvailable()
	return ErrHandoffQueueFull
}

// shutdownWorkers gracefully shuts down the worker manager, waiting for workers to complete
func ( *handoffWorkerManager) ( context.Context) error {
	.shutdownOnce.Do(func() {
		close(.shutdown)
		// workers will exit when they finish their current request

		// Shutdown circuit breaker manager cleanup goroutine
		if .circuitBreakerManager != nil {
			.circuitBreakerManager.Shutdown()
		}
	})

	// Wait for workers to complete
	 := make(chan struct{})
	go func() {
		.workerWg.Wait()
		close()
	}()

	select {
	case <-:
		return nil
	case <-.Done():
		return .Err()
	}
}
