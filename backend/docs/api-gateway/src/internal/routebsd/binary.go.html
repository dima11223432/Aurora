type binaryByteOrder interface {
	Uint16([]byte) uint16
	Uint32([]byte) uint32
	Uint64([]byte) uint64
}

type binaryLittleEndian struct{}

func (binaryLittleEndian) ( []byte) uint16 {
	return byteorder.LEUint16()
}

func (binaryLittleEndian) ( []byte) uint32 {
	return byteorder.LEUint32()
}

func (binaryLittleEndian) ( []byte) uint64 {
	return byteorder.LEUint64()
}

type binaryBigEndian struct{}

func (binaryBigEndian) ( []byte) uint16 {
	return byteorder.BEUint16()
}

func (binaryBigEndian) ( []byte) uint32 {
	return byteorder.BEUint32()
}

func (binaryBigEndian) ( []byte) uint64 {
	return byteorder.BEUint64()
}
